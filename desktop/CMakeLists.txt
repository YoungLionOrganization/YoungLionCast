function(add_cast_keychain)
  set(BUILD_TESTING OFF)
  set(BUILD_WITH_QT6 ON)
  set(BUILD_TEST_APPLICATION OFF)
  set(BUILD_TRANSLATIONS OFF)
  set(BUILD_SHARED_LIBS OFF)
  if(UNIX AND NOT APPLE)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(CAST_LIBSECRET QUIET libsecret-1)
    set(LIBSECRET_SUPPORT ${CAST_LIBSECRET_FOUND})
  endif()
  add_subdirectory(../third_party/qtkeychain qtkeychain EXCLUDE_FROM_ALL)
endfunction()
add_cast_keychain()
